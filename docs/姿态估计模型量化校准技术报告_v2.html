1.2 量化类型

🔧 训练后量化 (PTQ)

  • 不需要重新训练
  • 需要校准数据集（100-2000张）
  • 适合快速部署
  • 精度损失：1-3%
  • 推荐场景：大多数生产环境

🎯 量化感知训练 (QAT)

  • 需要重新训练（或微调）
  • 不需要校准数据
  • 适合高精度需求
  • 精度损失：<1%
  • 推荐场景：精度敏感应用

⚡ 动态量化

  • 不需要校准数据
  • 运行时动态计算量化参数
  • 精度损失：2-5%
  • 速度提升有限
  • 推荐场景：快速验证

1.3 量化数学原理

# 量化公式
量化 (Quantize):
    q = round(r / scale + zero_point)
    q = clamp(q, quant_min, quant_max)

反量化 (Dequantize):
    r = scale * (q - zero_point)

其中:
    r = 原始浮点值 (FP32)
    q = 量化后的整数值 (INT8)
    scale = 缩放因子 = (r_max - r_min) / (quant_max - quant_min)
    zero_point = 零点 = round(-r_min / scale)
    quant_min = -128 (INT8)
    quant_max = 127 (INT8)

  1. 硬件平台是什么？
    • NVIDIA GPU/Jetson → TensorRT

    • 有 → 静态量化
    • 没有 → 动态量化或 QAT

5.2 模型剪枝（Pruning）

移除不重要的权重或通道，减小模型大小。

# PyTorch 剪枝示例
import torch.nn.utils.prune as prune

# 结构化剪枝（移除整个通道）
for name, module in model.named_modules():
    if isinstance(module, torch.nn.Conv2d):
        prune.ln_structured(
            module, 
            name='weight', 
            amount=0.3,  # 剪枝30%
            n=2, 
            dim=0
        )

# 非结构化剪枝（移除单个权重）
prune.global_unstructured(
    parameters_to_prune,
    pruning_method=prune.L1Unstructured,
    amount=0.3
)

5.3 知识蒸馏（Knowledge Distillation）

用大模型（教师）指导小模型（学生）训练，提升量化后精度。

# 知识蒸馏示例
def distillation_loss(student_output, teacher_output, labels, temperature=4.0, alpha=0.7):
    """
    知识蒸馏损失
    - student_output: 学生模型输出
    - teacher_output: 教师模型输出
    - temperature: 温度参数（越高越平滑）
    - alpha: 蒸馏损失权重
    """
    # 软标签损失（KL散度）
    soft_loss = F.kl_div(
        F.log_softmax(student_output / temperature, dim=1),
        F.softmax(teacher_output / temperature, dim=1),
        reduction='batchmean'
    ) * (temperature ** 2)
    
    # 硬标签损失（交叉熵）
    hard_loss = F.cross_entropy(student_output, labels)
    
    # 总损失
    return alpha * soft_loss + (1 - alpha) * hard_loss

6.1 难点一：精度损失

⚠️
问题描述：姿态估计对精度要求高（关键点定位误差通常要求 < 5%），INT8 量化会损失信息，可能导致关键点偏移。
方案 描述 精度恢复 复杂度 推荐场景
混合精度量化 敏感层保持FP16/FP32 90-95% 中 通用场景
逐层量化 对每层单独优化量化策略 85-90% 高 精度敏感
量化感知训练 训练时模拟量化效果 95-98% 高 高精度需求
知识蒸馏 用FP32模型指导INT8模型 90-95% 高 大模型压缩

问题 可能原因 解决方案

def compare_outputs(fp32_model, int8_model, test_input):
    """对比量化前后模型输出"""
    fp32_output = fp32_model(test_input)
    int8_output = int8_model(test_input)

    cosine_sim = F.cosine_similarity(

        dim=0
    )
    
    print(f"MSE: {mse.item():.6f}")
    print(f"Cosine Similarity: {cosine_sim.item():.4f}")
    
    # 找出差异最大的位置
    diff = torch.abs(fp32_output - int8_output)
    max_diff_idx = torch.argmax(diff)
    print(f"最大差异位置: {max_diff_idx}")
    print(f"最大差异值: {diff.max().item():.6f}")

    plt.savefig(f"activation_{layer_name}.png")

def layerwise_accuracy_analysis(fp32_model, int8_model, val_data):
    """逐层分析精度下降"""
    results = []
    
    for layer_name in fp32_model.layers:
        # 仅量化该层
        model_partial = quantize_single_layer(fp32_model, layer_name)
        accuracy = evaluate(model_partial, val_data)
        
        fp32_accuracy = evaluate(fp32_model, val_data)
        drop = fp32_accuracy - accuracy
        
        results.append({
            'layer': layer_name,
            'accuracy': accuracy,
            'drop': drop
        })
    
    # 按精度下降排序
    results.sort(key=lambda x: x['drop'], reverse=True)
    
    print("逐层精度分析（按影响排序）：")
    for r in results[:10]:
        print(f"{r['layer']}: {r['accuracy']:.4f} (下降 {r['drop']:.4f})")

import tensorrt as trt
import numpy as np

class PoseEstimationCalibrator(trt.IInt8EntropyCalibrator2):
    """姿态估计模型校准器"""
    
    def __init__(self, calibration_dir, batch_size=8, input_shape=(1, 3, 256, 192)):
        super().__init__()
        self.calibration_dir = calibration_dir
        self.batch_size = batch_size
        self.input_shape = input_shape
        
        # 加载校准图片列表
        self.image_files = self.load_image_files(calibration_dir)
        
        # 随机选择校准样本（推荐500-2000张）
        if len(self.image_files) > 2000:
            self.image_files = np.random.choice(
                self.image_files, 2000, replace=False
            ).tolist()
        
        self.current_index = 0
    
    def get_batch(self, names):
        """获取下一个校准批次"""
        if self.current_index >= len(self.image_files):
            return None
        
        # 加载和预处理图片
        batch = []
        for i in range(self.batch_size):
            if self.current_index >= len(self.image_files):
                break
            
            img_path = self.image_files[self.current_index]
            img = self.preprocess_image(img_path)
            batch.append(img)
            self.current_index += 1
        
        if not batch:
            return None
        
        return np.array(batch, dtype=np.float32)


def quantize_pose_estimation_model(
    onnx_model_path,
    calibration_dir,
    output_engine_path,
    precision="INT8"
):
    """量化姿态估计模型"""
    logger = trt.Logger(trt.Logger.WARNING)
    builder = trt.Builder(logger)
    network = builder.create_network(
        1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
    )
    parser = trt.OnnxParser(network, logger)
    
    # 解析ONNX模型
    with open(onnx_model_path, 'rb') as model:
        if not parser.parse(model.read()):
            for error in range(parser.num_errors):
                print(parser.get_error(error))
            return None
    
    # 配置builder
    config = builder.create_builder_config()
    config.max_workspace_size = 1 << 30  # 1GB
    
    # 设置精度
    if precision == "INT8":
        config.set_flag(trt.BuilderFlag.INT8)
        calibrator = PoseEstimationCalibrator(
            calibration_dir=calibration_dir,
            batch_size=8
        )
        config.int8_calibrator = calibrator
    
    # 构建引擎
    engine = builder.build_serialized_network(network, config)
    
    # 保存引擎
    with open(output_engine_path, 'wb') as f:
        f.write(engine)
    
    return engine

import numpy as np
from pycocotools.coco import COCO
from pycocotools.cocoeval import COCOeval

def evaluate_quantized_model(
    fp32_results_file,
    int8_results_file,
    coco_annotation_file
):
    """评估量化前后模型精度"""
    # 加载COCO标注
    coco_gt = COCO(coco_annotation_file)
    
    # 加载FP32结果
    coco_dt_fp32 = coco_gt.loadRes(fp32_results_file)
    coco_eval_fp32 = COCOeval(coco_gt, coco_dt_fp32, 'keypoints')
    coco_eval_fp32.evaluate()
    coco_eval_fp32.accumulate()
    coco_eval_fp32.summarize()
    
    # 加载INT8结果
    coco_dt_int8 = coco_gt.loadRes(int8_results_file)
    coco_eval_int8 = COCOeval(coco_gt, coco_dt_int8, 'keypoints')
    coco_eval_int8.evaluate()
    coco_eval_int8.accumulate()
    coco_eval_int8.summarize()
    
    # 计算精度下降
    fp32_ap = coco_eval_fp32.stats[0]
    int8_ap = coco_eval_int8.stats[0]
    ap_drop = (fp32_ap - int8_ap) / fp32_ap * 100
    
    print(f"\n=== 精度对比 ===")
    print(f"FP32 AP: {fp32_ap:.4f}")
    print(f"INT8 AP: {int8_ap:.4f}")
    print(f"精度下降: {ap_drop:.2f}%")
    
    if ap_drop < 1:
        print("✅ 量化效果优秀")
    elif ap_drop < 3:
        print("✅ 量化效果良好")
    elif ap_drop < 5:
        print("⚠️ 量化效果一般，建议优化")
    else:
        print("❌ 量化效果不佳，需要重新调整")
    
    return {
        "fp32_ap": fp32_ap,
        "int8_ap": int8_ap,
        "ap_drop": ap_drop
    }
